cmake_minimum_required(VERSION 3.22)
project(svo_ros)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rclcpp_components REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(svo_msgs REQUIRED)
find_package(svo REQUIRED)
find_package(svo_cmake REQUIRED)
find_package(svo_ceres_backend REQUIRED)
find_package(svo_online_loopclosing REQUIRED)
find_package(svo_global_map REQUIRED)
find_package(vikit_ros REQUIRED)
find_package(rpg_common REQUIRED)
include(SvoSetup)
find_package(PCL REQUIRED COMPONENTS common)
add_library(${PROJECT_NAME} SHARED src/svo_interface.cpp src/svo_factory.cpp src/ceres_backend_factory.cpp src/svo_node_base.cpp src/visualizer.cpp src/csv_dataset_reader.cpp src/ego_input.cpp)
svo_target(${PROJECT_NAME})
ament_target_dependencies(${PROJECT_NAME} PUBLIC rclcpp rclcpp_components ament_index_cpp cv_bridge sensor_msgs geometry_msgs nav_msgs std_msgs visualization_msgs tf2_ros pcl_conversions svo_msgs svo svo_cmake svo_ceres_backend svo_online_loopclosing svo_global_map vikit_ros rpg_common)
target_include_directories(${PROJECT_NAME} SYSTEM PUBLIC ${PCL_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} PUBLIC ${PCL_LIBRARIES})
rclcpp_components_register_nodes(${PROJECT_NAME} "svo_ros::SvoNodeBase" "svo_ros::EgoInput")
add_executable(svo_node src/svo_node.cpp)
target_link_libraries(svo_node PRIVATE ${PROJECT_NAME})
install(TARGETS svo_node DESTINATION lib/${PROJECT_NAME})
add_executable(svo_benchmark src/benchmark_node.cpp)
target_link_libraries(svo_benchmark PRIVATE ${PROJECT_NAME})
install(TARGETS svo_benchmark DESTINATION lib/${PROJECT_NAME})
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY launch param DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(rclcpp rclcpp_components ament_index_cpp cv_bridge sensor_msgs geometry_msgs nav_msgs std_msgs visualization_msgs tf2_ros pcl_conversions svo_msgs svo svo_cmake svo_ceres_backend svo_online_loopclosing svo_global_map vikit_ros rpg_common PCL)
add_executable(ego_input_node src/ego_input_node.cpp)
target_link_libraries(ego_input_node PRIVATE ${PROJECT_NAME})
install(TARGETS ego_input_node DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY scripts/ DESTINATION share/${PROJECT_NAME}/scripts)
install(FILES rviz_config.rviz rviz_config_vio.rviz rviz_config_gm.rviz rqt.perspective PARAMETER_MIGRATION.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_input_scheduler test/test_input_scheduler.cpp)
  target_include_directories(test_input_scheduler PRIVATE include)
  ament_add_gtest(test_ego_math test/test_ego_math.cpp)
  target_include_directories(test_ego_math PRIVATE include)
  target_link_libraries(test_ego_math Eigen3::Eigen)
  ament_add_gtest(test_ego_component test/test_ego_component.cpp TIMEOUT 60)
  target_link_libraries(test_ego_component ${PROJECT_NAME})
  ament_add_gtest(test_runtime_lifecycle test/test_runtime_lifecycle.cpp TIMEOUT 60)
  target_link_libraries(test_runtime_lifecycle ${PROJECT_NAME})
endif()
ament_package()
